```tsx
import { useEffect, useState } from 'react';
import { PhaserGame } from './components/PhaserGame';
import { GameUI } from './components/GameUI';
import { MainMenu } from './components/MainMenu';
import { Shop } from './components/Shop';
import { Settings } from './components/Settings';
import { Leaderboard } from './components/Leaderboard';
import { Login } from './components/Login';
import { useGameStore } from './store/useGameStore';
import { useUserStore } from './store/useUserStore';
import { supabase } from './lib/supabase';

function App() {
  const { gameState } = useGameStore();
  const { setUser } = useUserStore();
  const [init, setInit] = useState(false);

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      if (session?.user) {
        supabase.from('profiles').select('*').eq('id', session.user.id).single().then(({ data }) => {
          if (data) {
            setUser({
              id: data.id,
              role: data.role,
              username: data.username,
              heatPoints: data.heat_points,
              crystals: data.crystals,
              bestWave: data.best_wave,
              bestScore: data.best_score,
              totalGames: data.total_games,
              upgrades: {
                startTemp: data.upgrades.start_temp || 0,
                heatSpeed: data.upgrades.heat_speed || 0,
                coolingResist: data.upgrades.cooling_resist || 0,
                fuelCap: data.upgrades.fuel_cap || 0,
                blastRadius: data.upgrades.blast_radius || 0,
              },
              skins: data.skins,
              currentSkin: data.current_skin,
            });
          }
          setInit(true);
        });
      } else {
        setInit(true);
      }
    });

    supabase.auth.onAuthStateChange((_event, session) => {
      if (!session) {
        setUser({ id: null, role: 'user', username: '', heatPoints: 0, crystals: 0 });
      }
    });
  }, []);

  if (!init) return <div className="bg-[#0A0A1A] w-screen h-screen flex items-center justify-center text-white text-2xl font-bold">LOADING...</div>;

  return (
    <div className="w-screen h-screen overflow-hidden bg-[#0A0A1A] relative select-none">
      <PhaserGame />
      <GameUI />
      {gameState === 'menu' && <MainMenu />}
      {gameState === 'shop' && <Shop />}
      {gameState === 'settings' && <Settings />}
      {gameState === 'leaderboard' && <Leaderboard />}
      {gameState === 'login' && <Login />}
    </div>
  );
}

export default App;
```

```ts
import { create } from 'zustand';

interface GameState {
  score: number;
  wave: number;
  combo: number;
  gameState: 'menu' | 'playing' | 'paused' | 'gameover' | 'shop' | 'leaderboard' | 'settings' | 'login';
  setGameState: (state: GameState['gameState']) => void;
  addScore: (points: number) => void;
  setWave: (wave: number) => void;
  setCombo: (combo: number) => void;
  resetGame: () => void;
}

export const useGameStore = create<GameState>((set) => ({
  score: 0,
  wave: 1,
  combo: 1,
  gameState: 'menu',
  setGameState: (state) => set({ gameState: state }),
  addScore: (points) => set((state) => ({ score: state.score + points * state.combo })),
  setWave: (wave) => set({ wave }),
  setCombo: (combo) => set({ combo }),
  resetGame: () => set({ score: 0, wave: 1, combo: 1 }),
}));
```

```ts
import { create } from 'zustand';

interface UserState {
  id: string | null;
  username: string;
  role: 'user' | 'admin';
  heatPoints: number;
  crystals: number;
  bestWave: number;
  bestScore: number;
  totalGames: number;
  upgrades: {
    startTemp: number; // 0-5
    heatSpeed: number; // 0-5
    coolingResist: number; // 0-5
    fuelCap: number; // 0-3
    blastRadius: number; // 0-3
  };
  skins: string[];
  currentSkin: string;
  setUser: (data: Partial<UserState>) => void;
  addHeatPoints: (points: number) => void;
  addCrystals: (amount: number) => void;
  updateBest: (wave: number, score: number) => void;
}

export const useUserStore = create<UserState>((set) => ({
  id: null,
  username: '',
  role: 'user',
  heatPoints: 0,
  crystals: 0,
  bestWave: 0,
  bestScore: 0,
  totalGames: 0,
  upgrades: {
    startTemp: 0,
    heatSpeed: 0,
    coolingResist: 0,
    fuelCap: 0,
    blastRadius: 0
  },
  skins: ['default'],
  currentSkin: 'default',
  setUser: (data) => set((state) => ({ ...state, ...data })),
  addHeatPoints: (points) => set((state) => ({ heatPoints: state.heatPoints + points })),
  addCrystals: (amount) => set((state) => ({ crystals: state.crystals + amount })),
  updateBest: (wave, score) => set((state) => ({
    bestWave: Math.max(state.bestWave, wave),
    bestScore: Math.max(state.bestScore, score),
  })),
}));
```

```tsx
import { useEffect } from 'react';
import { useGameStore } from '../store/useGameStore';
import { useUserStore } from '../store/useUserStore';
import { supabase } from '../lib/supabase';
import { Flame, Droplet, Settings, ShoppingCart, Play, Zap } from 'lucide-react';
import { EventBus } from '../game/EventBus';

export function GameUI() {
  const { gameState, setGameState, score, wave, combo } = useGameStore();
  const { id, bestScore, bestWave, heatPoints, crystals } = useUserStore();

  useEffect(() => {
    EventBus.on('game-over', () => {
      setGameState('gameover');
      // Save score to supabase if logged in
      if (id) {
        const storeScore = useGameStore.getState().score;
        const storeWave = useGameStore.getState().wave;
        
        supabase.from('leaderboards').insert({
          user_id: id,
          score: storeScore,
          wave: storeWave,
        }).then();

        // Update profile
        const uState = useUserStore.getState();
        supabase.from('profiles').update({
          best_score: Math.max(uState.bestScore, storeScore),
          best_wave: Math.max(uState.bestWave, storeWave),
          heat_points: uState.heatPoints, // Already added in MainGame.ts
        }).eq('id', id).then();
      }
    });
    return () => {
      EventBus.off('game-over');
    };
  }, [id, setGameState]);

  if (gameState === 'playing') {
    return (
      <div className="absolute top-0 left-0 w-full p-4 pointer-events-none flex justify-between items-start text-white">
        <div className="flex flex-col gap-1">
          <div className="text-xl font-bold uppercase tracking-wider text-[#FFD700]">Wave {wave}</div>
          <div className="text-3xl font-bold">{score}</div>
        </div>
        {combo > 1 && (
          <div className="flex flex-col items-end">
            <div className="text-2xl font-bold text-[#FF8800] animate-bounce">
              {combo}x COMBO!
            </div>
          </div>
        )}
      </div>
    );
  }

  if (gameState === 'gameover') {
    return (
      <div className="absolute inset-0 bg-black/80 flex flex-col items-center justify-center text-white pointer-events-auto p-6">
        <h2 className="text-4xl font-bold mb-2 text-red-500">GAME OVER</h2>
        <div className="text-xl mb-8 flex flex-col items-center gap-2">
          <div>Score: <span className="font-bold text-[#FFD700]">{score}</span></div>
          <div>Wave Reached: <span className="font-bold text-[#FF8800]">{wave}</span></div>
          <div className="text-gray-400 text-sm mt-4">Best: {bestScore} (Wave {bestWave})</div>
        </div>
        
        <div className="flex flex-col gap-4 w-full max-w-xs">
          <button 
            onClick={() => { 
              if (crystals >= 50) {
                // Deduct crystals
                useUserStore.getState().addCrystals(-50);
                if (id) {
                  supabase.from('profiles').update({ crystals: crystals - 50 }).eq('id', id).then();
                }
                setGameState('playing'); 
                EventBus.emit('game-continue');
              } else {
                alert('Not enough crystals to continue! Need 50.');
              }
            }} 
            className="bg-[#0066FF] text-white py-3 rounded-xl font-bold text-lg hover:bg-blue-500 transition-colors flex items-center justify-center gap-2"
          >
            CONTINUE <span className="text-sm">(50 💎)</span>
          </button>
          <button 
            onClick={() => { setGameState('playing'); EventBus.emit('scene-changed', { key: 'MainGame' }); }} 
            className="bg-[#FF8800] text-black py-3 rounded-xl font-bold text-lg hover:bg-[#FFD700] transition-colors"
          >
            RESTART
          </button>
          <button 
            onClick={() => setGameState('menu')} 
            className="bg-gray-800 text-white py-3 rounded-xl font-bold hover:bg-gray-700 transition-colors"
          >
            MAIN MENU
          </button>
        </div>
      </div>
    );
  }

  return null; // For Menu, Shop, Settings, they will be handled by App.tsx or separate components
}
```

```tsx
import { useEffect, useState } from 'react';
import { useGameStore } from '../store/useGameStore';
import { ArrowLeft, Trophy } from 'lucide-react';
import { supabase } from '../lib/supabase';

interface LeaderboardEntry {
  id: string;
  username: string;
  best_score: number;
  best_wave: number;
}

export function Leaderboard() {
  const { setGameState } = useGameStore();
  const [entries, setEntries] = useState<LeaderboardEntry[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function load() {
      const { data } = await supabase
        .from('public_profiles')
        .select('*')
        .order('best_score', { ascending: false })
        .limit(100);
      
      if (data) setEntries(data);
      setLoading(false);
    }
    load();
  }, []);

  return (
    <div className="absolute inset-0 bg-[#0A0A1A] z-50 text-white flex flex-col p-4">
      <div className="flex items-center justify-between mb-6 pt-4 px-2">
        <button onClick={() => setGameState('menu')} className="p-2 hover:bg-gray-800 rounded-full">
          <ArrowLeft size={28} />
        </button>
        <h2 className="text-2xl font-bold uppercase tracking-wider text-[#FFD700] flex items-center gap-2">
          <Trophy size={24} /> LEADERBOARD
        </h2>
        <div className="w-10" />
      </div>

      <div className="flex-1 overflow-y-auto px-2 pb-8">
        {loading ? (
          <div className="text-center text-gray-500 mt-10">Loading...</div>
        ) : (
          <div className="space-y-2">
            {entries.map((e, i) => (
              <div key={e.id} className="flex items-center justify-between bg-gray-900 border border-gray-800 p-4 rounded-xl">
                <div className="flex items-center gap-4">
                  <div className={`w-8 text-center font-bold ${i === 0 ? 'text-[#FFD700]' : i === 1 ? 'text-gray-300' : i === 2 ? 'text-amber-600' : 'text-gray-500'}`}>
                    #{i + 1}
                  </div>
                  <div className="font-medium text-lg">{e.username || 'Anonymous'}</div>
                </div>
                <div className="text-right">
                  <div className="font-bold text-[#FFD700]">{e.best_score}</div>
                  <div className="text-xs text-[#FF8800]">Wave {e.best_wave}</div>
                </div>
              </div>
            ))}
            {entries.length === 0 && (
              <div className="text-center text-gray-500 mt-10">No entries yet.</div>
            )}
          </div>
        )}
      </div>
    </div>
  );
}
```

```tsx
import { useState } from 'react';
import { supabase } from '../lib/supabase';
import { useGameStore } from '../store/useGameStore';
import { useUserStore } from '../store/useUserStore';
import { ArrowLeft } from 'lucide-react';

export function Login() {
  const { setGameState } = useGameStore();
  const { setUser } = useUserStore();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [username, setUsername] = useState('');
  const [isRegister, setIsRegister] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [agreed, setAgreed] = useState(false);

  const handleAuth = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setError('');

    try {
      if (isRegister) {
        if (!agreed) {
          throw new Error('You must agree to the Terms of Service and Privacy Policy');
        }
        if (!username) {
          throw new Error('Username is required');
        }
        const { data, error: err } = await supabase.auth.signUp({
          email: `${username}@miaoda.com`, // using username to simulate
          password,
        });
        if (err) throw err;
        
        // Update profile username right after signup since we just registered
        if (data.user) {
          await supabase.from('profiles').update({ username }).eq('id', data.user.id);
          setUser({ id: data.user.id, username });
        }
        
        setGameState('menu');
      } else {
        const { data, error: err } = await supabase.auth.signInWithPassword({
          email: `${username}@miaoda.com`, // treat email field as username field basically
          password,
        });
        if (err) throw err;
        
        if (data.user) {
          // Load profile
          const { data: profile } = await supabase.from('profiles').select('*').eq('id', data.user.id).single();
          if (profile) {
            setUser({
              id: profile.id,
              role: profile.role,
              username: profile.username,
              heatPoints: profile.heat_points,
              crystals: profile.crystals,
              bestWave: profile.best_wave,
              bestScore: profile.best_score,
              totalGames: profile.total_games,
              upgrades: {
                startTemp: profile.upgrades.start_temp || 0,
                heatSpeed: profile.upgrades.heat_speed || 0,
                coolingResist: profile.upgrades.cooling_resist || 0,
                fuelCap: profile.upgrades.fuel_cap || 0,
                blastRadius: profile.upgrades.blast_radius || 0,
              },
              skins: profile.skins,
              currentSkin: profile.current_skin,
            });
          }
        }
        setGameState('menu');
      }
    } catch (err: any) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="absolute inset-0 bg-[#0A0A1A] z-50 text-white flex flex-col p-4 items-center justify-center">
      <div className="absolute top-8 left-6">
        <button onClick={() => setGameState('menu')} className="p-2 hover:bg-gray-800 rounded-full">
          <ArrowLeft size={28} />
        </button>
      </div>

      <div className="bg-gray-900 border border-gray-800 p-8 rounded-2xl w-full max-w-sm shadow-2xl">
        <h2 className="text-3xl font-bold uppercase tracking-wider text-[#FFD700] mb-8 text-center">
          {isRegister ? 'REGISTER' : 'LOGIN'}
        </h2>

        {error && <div className="bg-red-900/50 border border-red-500 text-red-200 p-3 rounded-lg text-sm mb-6">{error}</div>}

        <form onSubmit={handleAuth} className="space-y-4">
          <div>
            <label className="block text-sm text-gray-400 mb-1">Username</label>
            <input 
              type="text" 
              required
              value={username}
              onChange={e => setUsername(e.target.value)}
              className="w-full bg-gray-800 border border-gray-700 rounded-lg px-4 py-3 focus:border-[#FFD700] outline-none"
              placeholder="Enter username"
            />
          </div>
          <div>
            <label className="block text-sm text-gray-400 mb-1">Password</label>
            <input 
              type="password" 
              required
              value={password}
              onChange={e => setPassword(e.target.value)}
              className="w-full bg-gray-800 border border-gray-700 rounded-lg px-4 py-3 focus:border-[#FFD700] outline-none"
              placeholder="Enter password"
            />
          </div>

          {isRegister && (
            <label className="flex items-start gap-2 text-sm text-gray-400 cursor-pointer mt-4">
              <input 
                type="checkbox" 
                checked={agreed}
                onChange={e => setAgreed(e.target.checked)}
                className="mt-1"
              />
              <span>I agree to the Terms of Service and Privacy Policy</span>
            </label>
          )}

          <button 
            type="submit" 
            disabled={loading}
            className="w-full bg-[#FF8800] text-black py-4 rounded-xl font-bold text-lg hover:bg-[#FFD700] transition-colors mt-8 disabled:opacity-50"
          >
            {loading ? 'PLEASE WAIT...' : isRegister ? 'CREATE ACCOUNT' : 'LOGIN'}
          </button>
        </form>

        <div className="mt-8 text-center text-gray-400 text-sm">
          {isRegister ? 'Already have an account?' : 'Need an account?'}

            className="ml-2 text-[#FFD700] hover:underline font-bold"
          >
            {isRegister ? 'Login here' : 'Register here'}
          </button>
        </div>

  );
}
```

```tsx
import { useGameStore } from '../store/useGameStore';
import { useUserStore } from '../store/useUserStore';
import { ArrowLeft, Flame, Zap, Shield, Battery, Bomb, ShoppingCart, Gem } from 'lucide-react';
import { useState } from 'react';
import { supabase } from '../lib/supabase';

const UPGRADES = [
  { key: 'startTemp', name: 'Start Temp', icon: <Flame size={20} color="#FF8800" />, desc: '+5° spawn temp', costs: [500, 1000, 2000, 4000, 8000] },
  { key: 'heatSpeed', name: 'Heat Speed', icon: <Zap size={20} color="#FFD700" />, desc: '+3° per tap', costs: [300, 600, 1200, 2400, 4800] },
  { key: 'coolingResist', name: 'Heat Retain', icon: <Shield size={20} color="#0066FF" />, desc: '-4% cooling rate', costs: [400, 800, 1600, 3200, 6400] },
  { key: 'fuelCap', name: 'Fuel Cap', icon: <Battery size={20} color="#00FF00" />, desc: '+1 max fuel', costs: [1000, 2500, 5000] },
  { key: 'blastRadius', name: 'Blast Radius', icon: <Bomb size={20} color="#FF1100" />, desc: 'Hot ignites neighbors', costs: [2000, 5000, 10000] },
];

const SKINS = [
  { id: 'default', name: 'Default', price: 0 },
  { id: 'magma', name: 'Magma', price: 200, isPremium: true },
  { id: 'ice', name: 'Ice Block', price: 200, isPremium: true },
  { id: 'neon', name: 'Neon Frame', price: 400, isPremium: true }
];

export function Shop() {
  const { setGameState } = useGameStore();
  const { heatPoints, upgrades, crystals, skins, currentSkin, setUser, id } = useUserStore();
  const [tab, setTab] = useState<'upgrades' | 'skins' | 'premium'>('upgrades');

  const buyUpgrade = (key: keyof typeof upgrades) => {
    const level = upgrades[key];
    const def = UPGRADES.find(u => u.key === key);
    if (!def || level >= def.costs.length) return;
    
    const cost = def.costs[level];
    if (heatPoints >= cost) {
      const newUpgrades = { ...upgrades, [key]: level + 1 };
      setUser({ 
        heatPoints: heatPoints - cost, 
        upgrades: newUpgrades 
      });
      if (id) {
        const dbUpgrades = {
          start_temp: newUpgrades.startTemp,
          heat_speed: newUpgrades.heatSpeed,
          cooling_resist: newUpgrades.coolingResist,
          fuel_cap: newUpgrades.fuelCap,
          blast_radius: newUpgrades.blastRadius,
        };
        supabase.from('profiles').update({ 
          heat_points: heatPoints - cost, 
          upgrades: dbUpgrades 
        }).eq('id', id).then();
      }
    }
  };

  const buySkin = (skinId: string, price: number) => {
    if (skins.includes(skinId)) {
      setUser({ currentSkin: skinId });
      if (id) supabase.from('profiles').update({ current_skin: skinId }).eq('id', id).then();
    } else if (crystals >= price) {
      const newSkins = [...skins, skinId];
      setUser({ 
        crystals: crystals - price,
        skins: newSkins,
        currentSkin: skinId 
      });
      if (id) {
        supabase.from('profiles').update({ 
          crystals: crystals - price,
          skins: newSkins,
          current_skin: skinId
        }).eq('id', id).then();
      }
    }
  };

  const checkoutCrystals = async (sku: string, amount: number, price: number) => {
    try {
      const { data: sessionData, error } = await supabase.functions.invoke('create_stripe_checkout', {
        body: {
          items: [{ name: amount > 0 ? `${amount} Crystals` : 'Remove Ads', price, quantity: 1, type: amount > 0 ? 'crystals' : 'skin', sku }]
        }
      });
      if (error) throw error;
      if (sessionData && sessionData.url) {
        window.open(sessionData.url, '_blank');
      }
    } catch (e) {
      console.error(e);
      alert('Failed to start checkout. Check console.');
    }
  };

  return (
    <div className="absolute inset-0 bg-[#0A0A1A] z-50 text-white flex flex-col p-4">
      <div className="flex items-center justify-between mb-6 pt-4 px-2">
        <button onClick={() => setGameState('menu')} className="p-2 hover:bg-gray-800 rounded-full">
          <ArrowLeft size={28} />
        </button>
        <h2 className="text-2xl font-bold uppercase tracking-wider text-[#FFD700]">SHOP</h2>
        <div className="flex gap-4">
          <div className="flex items-center gap-1 font-bold"><Flame size={16} color="#FF8800"/> {heatPoints}</div>
          <div className="flex items-center gap-1 font-bold text-blue-300"><Gem size={16} /> {crystals}</div>
        </div>
      </div>

      <div className="flex gap-2 mb-6 px-2">
        <button onClick={() => setTab('upgrades')} className={`flex-1 py-2 rounded-lg font-bold text-sm ${tab === 'upgrades' ? 'bg-[#FF8800] text-black' : 'bg-gray-800'}`}>UPGRADES</button>
        <button onClick={() => setTab('skins')} className={`flex-1 py-2 rounded-lg font-bold text-sm ${tab === 'skins' ? 'bg-blue-600 text-white' : 'bg-gray-800'}`}>SKINS</button>
        <button onClick={() => setTab('premium')} className={`flex-1 py-2 rounded-lg font-bold text-sm ${tab === 'premium' ? 'bg-[#FFD700] text-black' : 'bg-gray-800'}`}>BUY 💎</button>
      </div>

      <div className="flex-1 overflow-y-auto px-2 pb-8 space-y-4">
        {tab === 'upgrades' && UPGRADES.map(u => {
          const level = upgrades[u.key as keyof typeof upgrades];
          const maxed = level >= u.costs.length;
          const cost = maxed ? 0 : u.costs[level];
          const canAfford = heatPoints >= cost;

          return (
            <div key={u.key} className="bg-gray-900 border border-gray-800 p-4 rounded-xl flex items-center justify-between">
              <div className="flex items-center gap-4">
                <div className="bg-gray-800 p-3 rounded-lg">{u.icon}</div>
                <div>
                  <div className="font-bold">{u.name}</div>
                  <div className="text-xs text-gray-400">{u.desc}</div>
                  <div className="text-xs text-[#FF8800] mt-1 font-mono">LVL {level}/{u.costs.length}</div>
                </div>
              </div>
              <button 
                onClick={() => buyUpgrade(u.key as any)}
                disabled={maxed || !canAfford}
                className={`px-4 py-2 rounded-lg font-bold flex items-center gap-1 ${maxed ? 'bg-gray-700 text-gray-500' : canAfford ? 'bg-[#FF8800] text-black hover:bg-[#FFD700]' : 'bg-gray-800 text-red-400'}`}
              >
                {maxed ? 'MAX' : <><Flame size={14}/> {cost}</>}
              </button>
            </div>
          );
        })}

        {tab === 'skins' && SKINS.map(s => {
          const owned = skins.includes(s.id);
          const active = currentSkin === s.id;
          
          return (
            <div key={s.id} className={`bg-gray-900 border ${active ? 'border-blue-500' : 'border-gray-800'} p-4 rounded-xl flex items-center justify-between`}>
              <div className="font-bold">{s.name}</div>
              <button 
                onClick={() => buySkin(s.id, s.price)}
                className={`px-4 py-2 rounded-lg font-bold flex items-center gap-1 ${active ? 'bg-blue-600 text-white' : owned ? 'bg-gray-700 text-white hover:bg-gray-600' : crystals >= s.price ? 'bg-blue-500 text-white hover:bg-blue-400' : 'bg-gray-800 text-red-400'}`}
              >
                {active ? 'EQUIPPED' : owned ? 'EQUIP' : <><Gem size={14}/> {s.price}</>}
              </button>
            </div>
          );
        })}

        {tab === 'premium' && (
          <>
            <div className="bg-gray-900 border border-blue-900 p-4 rounded-xl flex items-center justify-between">
              <div>
                <div className="font-bold text-blue-300">Handful of Crystals</div>
                <div className="text-sm text-gray-400">100 💎</div>
              </div>
              <button onClick={() => checkoutCrystals('crystals_100', 100, 0.99)} className="bg-blue-600 text-white px-4 py-2 rounded-lg font-bold hover:bg-blue-500">$0.99</button>
            </div>
            <div className="bg-gray-900 border border-blue-900 p-4 rounded-xl flex items-center justify-between">
              <div>
                <div className="font-bold text-blue-300">Pouch of Crystals</div>
                <div className="text-sm text-gray-400">500 💎</div>
              </div>
              <button onClick={() => checkoutCrystals('crystals_500', 500, 3.99)} className="bg-blue-600 text-white px-4 py-2 rounded-lg font-bold hover:bg-blue-500">$3.99</button>
            </div>
            <div className="bg-gray-900 border border-blue-900 p-4 rounded-xl flex items-center justify-between">
              <div>
                <div className="font-bold text-blue-300">Chest of Crystals</div>
                <div className="text-sm text-gray-400">2000 💎</div>
              </div>
              <button onClick={() => checkoutCrystals('crystals_2000', 2000, 9.99)} className="bg-blue-600 text-white px-4 py-2 rounded-lg font-bold hover:bg-blue-500">$9.99</button>

                <div className="font-bold text-[#FFD700]">Remove Ads</div>
                <div className="text-sm text-gray-400">Permanent upgrade</div>
              </div>

            </div>
          </>
        )}
      </div>
    </div>
  );
}
```
